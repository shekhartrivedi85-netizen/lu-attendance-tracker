function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Daily Log (P1)") || ss.getSheets()[0]; const data = sheet.getDataRange().getValues(); // Row 4 se data start hota hai (0-indexed slice 3) const rows = data.slice(3); const result = []; for (let i = 0; i < rows.length; i++) { const r = rows[i]; if (!r[1] && !r[3]) continue; // Blank row skip let dateIso = ""; if (r[1] instanceof Date) { dateIso = Utilities.formatDate(r[1], "Asia/Kolkata", "yyyy-MM-dd"); } else if (r[1]) { const d = new Date(r[1]); if (!isNaN(d.getTime())) { dateIso = Utilities.formatDate(d, "Asia/Kolkata", "yyyy-MM-dd"); } else { dateIso = String(r[1]).trim(); } } result.push({ rowIndex: i + 4, sno: r[0], date: dateIso, // Hamesha standard YYYY-MM-DD rahega day: String(r[2] || "").trim(), slot: String(r[3] || "").trim(), code: String(r[4] || "").trim(), category: String(r[5] || "").trim(), subject: String(r[6] || "").trim(), status: r[7] ? String(r[7]).trim().toUpperCase() : "" }); } return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { try { let postData; if (e.postData && e.postData.contents) { postData = JSON.parse(e.postData.contents); } else { postData = e.parameter; } const { date, slot, code, status } = postData; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Daily Log (P1)") || ss.getSheets()[0]; const data = sheet.getDataRange().getValues(); let updated = false; for (let i = 3; i < data.length; i++) { let rowDateIso = ""; if (data[i][1] instanceof Date) { rowDateIso = Utilities.formatDate(data[i][1], "Asia/Kolkata", "yyyy-MM-dd"); } else if (data[i][1]) { const d = new Date(data[i][1]); rowDateIso = !isNaN(d.getTime()) ? Utilities.formatDate(d, "Asia/Kolkata", "yyyy-MM-dd") : String(data[i][1]).trim(); } const rowSlot = String(data[i][3]).trim(); const rowCode = String(data[i][4]).trim(); if (rowDateIso === date && rowSlot === slot && rowCode === code) { sheet.getRange(i + 1, 8).setValue(status || ""); // Column H (Status) updated = true; break; } } return ContentService.createTextOutput(JSON.stringify({ success: updated })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
